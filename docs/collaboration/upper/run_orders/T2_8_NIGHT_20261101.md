```powershell
$py = "D:\py\DRL2\.venv\Scripts\python.exe"; $R = "C:\Users\35884\Documents\Spacecraft\过程文件\T2_8_NIGHT_20261101"
Set-Location "D:\py\DRL2"; $env:OMP_NUM_THREADS = "1"; $env:MKL_NUM_THREADS = "1"; $env:OPENBLAS_NUM_THREADS = "1"
$L = [int](& $py -c "import os; print(os.cpu_count())")
$T0 = Get-Date; $hours = { ((Get-Date) - $T0).TotalHours }
$W = if ($L -ge 24) { 3 } else { 2 }; $EW = [Math]::Max(4, $L - 2)
"start=$T0 W=$W EW=$EW" | Out-File -Encoding utf8 "$R\night.log"

# 1. P3 key-state scoring (skip if T2_8_KEYSCORE already delivered)
$Wd = "C:\Users\35884\Documents\Spacecraft\过程文件\T2_6W_WIDE_140_200\B"
$W1 = "C:\Users\35884\Documents\Spacecraft\过程文件\T2_6W1_CUTOFF_20261029\B"
$G  = "C:\Users\35884\Documents\Spacecraft\过程文件\T2_7_GUIDE_20261031\B"
$ck = @()
foreach ($r in @(@("e_sac","ng",$Wd), @("e_mve","ng",$W1), @("e_sac","g",$G), @("e_mve","g",$G))) { foreach ($s in 0,1) { foreach ($k in 1000,2000,3000,4000) {
  $z = "$($r[2])\TRAIN_$($r[0])_s$s\checkpoints\step_$k.zip"; if (Test-Path $z) { $ck += "$($r[0])_s$($s)_$($r[1])_$k=$z" } } } }
& $py -B -m experiments.t2_keyscore --states evidence/T2_8_PRECHECK/keystates --checkpoints $ck --output "$R\KEYSCORE.json" *> "$R\keyscore.log"
& $py -B evidence/T2_8_PRECHECK/脚本/keyscore_summary.py evidence/T2_8_PRECHECK/keystates "$R\KEYSCORE.json" "$R\KEYSCORE_SUMMARY.json" *> "$R\keyscore_summary.log"
"P3 done $(& $hours) h, checkpoints $($ck.Count)" | Out-File -Encoding utf8 -Append "$R\night.log"

# 2. training: 4 runs from zero, in parallel
$procs = @()
foreach ($arm in "e_mve","e_sac") { foreach ($s in 0,1) {
  $out = "$R\B\TRAIN_${arm}_s$s"
  $procs += Start-Process -FilePath $py -WorkingDirectory "D:\py\DRL2" -WindowStyle Hidden -PassThru `
    -ArgumentList "-B","-m","experiments.t2","train","--arm",$arm,"--seed",$s,"--steps","2000","--workers",$W,"--urgent-deadline","140,200",
                  "--checkpoints","500,1000,1500","--output",$out `
    -RedirectStandardOutput "$R\B\TRAIN_${arm}_s$s.out.log" -RedirectStandardError "$R\B\TRAIN_${arm}_s$s.err.log" } }

$runs = foreach ($arm in "e_mve","e_sac") { foreach ($s in 0,1) { "${arm}_s$s" } }

$top = @(2000, 1500, 1000) | Where-Object { & $have $_ } | Select-Object -First 1
$evaluated = @()
foreach ($k in (@($top, 1000, 1500, 500) | Where-Object { $_ } | Select-Object -Unique)) {
  if ((& $hours) -ge 10.75) { "eval stop before step_$k at $(& $hours) h" | Out-File -Encoding utf8 -Append "$R\night.log"; break }

      --openings 294000-294047 --urgent-deadline 140,200 --workers $EW --output "$R\C\$r\step_$k" *>> "$R\eval.log" }
  $evaluated += $k; "eval step_$k done $(& $hours) h" | Out-File -Encoding utf8 -Append "$R\night.log" }

# 4. readout and the pre-registered decision
& $py -B -m experiments.t2_readout curve --eval-root "$R\C" --train-root "$R\B" --screen "$R\A_none" --expected 48 --output "$R\READOUT.json" *> "$R\readout.log"
$dk = @(1000) + @(@(2000, 1500) | Where-Object { $evaluated -contains $_ } | Select-Object -First 1)
& $py -B -m experiments.t2_decide --criteria t28 --checkpoints (($dk | Where-Object { $_ }) -join ",") --readout "$R\READOUT.json" --train-root "$R\B" --output "$R\DECISION.json" *> "$R\decide.log"
"all done $(& $hours) h; decision checkpoints $($dk -join ',')" | Out-File -Encoding utf8 -Append "$R\night.log"
```

## 预先登记的判据（`t28`，事后不得修改）

**比较方式：** 每条训练的 AUC 为它在判定检查点上 48 个开局平均回报的均值；按臂比较，不按种子配对。判定检查点为 1000 加上已评估到的最高检查点（优先 2000，其次 1500）。

| 项 | 标准 |
|---|---|
| C 完整 | 四条训练在判定检查点上都是 48/48，且没有 ERROR 文件 |
| L 学习效果 | EACC 臂平均 AUC − SAC 臂平均 AUC ≥ +5 |
| R 稳定性 | EACC 最差一条的 AUC ≥ SAC 最差一条 + 5，且 EACC 至多一条不高于 SAC 臂均值 |
| S 安全 | EACC 臂 unsafe 总数 ≤ SAC 臂 + 1 |

**判定：**

| 结果 | 条件 |
|---|---|
| `pass` | C、L、R、S 全过 |
| `partial` | C、S 过，L 与 R 只过一项 |
| `harmful` | S 不过，或 EACC 臂均值比 SAC 低 5 以上 |
| `no_gain` | 其余情况 |
| `incomplete` | C 不过 |

**只报告、不设门：**
- 各检查点两臂的均值与离散；
- 紧急 / 经济捕获；
- 成功回合的燃料与时间；
- EACC 机制读数；
- 种子配对的差（仅供参考）；
- P3 的 critic 判对率与策略选中真值最优档的比例。

每臂 2 个种子，稳定性的读数有限，按探索性证据对待。
